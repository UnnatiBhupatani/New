<?php

namespace App\Models;

use Illuminate\Foundation\Auth\User as Authenticatable;

class Student extends Authenticatable
{
    // Add this method to check if the student has a specific permission
    public function hasPermission($permissionName)
    {
        // Fetch the permissions from the role associated with the student
        return $this->role->permissions()->where('name', $permissionName)->exists();
    }

    // Define the relationship to the role
    public function role()
    {
        return $this->belongsTo(Role::class);
    }
}

Route::get('/dashboard', function () {
    $user = auth()->user();

    // Check if user has 'view_dashboard' permission
    if ($user && $user->hasPermission('view_dashboard')) {
        return view('dashboard');
    }

    return redirect('/unauthorized')->with('error', 'You do not have permission to access this page.');
});

Route::get('/manage-students', function () {
    $user = auth()->user();

    // Only allow access to Admin role
    if ($user && $user->hasPermission('manage_students')) {
        return view('manage-students');
    }

    return redirect('/unauthorized')->with('error', 'You do not have permission to access this page.');
});
