class Role extends Model
{
    public function permissions()
    {
        return $this->belongsToMany(Permission::class, 'role_permission', 'role_id', 'permission_id');
    }
}


class Permission extends Model
{
    public function roles()
    {
        return $this->belongsToMany(Role::class, 'role_permission', 'permission_id', 'role_id');
    }
}




public function index()
{
    $roles = Role::with('permissions')->get(); // Fetch all roles with their permissions
    return view('roles.index', compact('roles'));
}

namespace App\Http\Controllers;

use App\Models\Permission;
use App\Models\Role;
use Illuminate\Http\Request;

class RolePermissionController extends Controller
{
    public function fetch_rolePermission(Request $request)
    {
        // Fetch all roles and permissions
        $roles = Role::all();
        $permissions = Permission::all();
        $role_permission = Role::with('permissions')->get();

        // Get selected role's permissions (default empty if no role is selected)
        $selectedRole = $request->input('role'); // Get role ID from request
        $checkedPermissions = [];

        if ($selectedRole) {
            $role = Role::find($selectedRole);
            if ($role) {
                $checkedPermissions = $role->permissions()->pluck('id')->toArray(); // Fetch assigned permissions
            }
        }

        return view('role_permission', compact('roles', 'permissions', 'role_permission', 'selectedRole', 'checkedPermissions'));
    }

    public function store(Request $request)
    {
        $role = Role::find($request->role);
        if ($role) {
            $role->permissions()->sync($request->permission ?? []);
        }

        return redirect('role_permission')->with('success', 'Permissions updated successfully.');
    }
}
